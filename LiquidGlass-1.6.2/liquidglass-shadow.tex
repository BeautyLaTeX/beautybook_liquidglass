% All shadow paint is outside the native frame. Its centre is not added
% to the body, and open page-cut edges are clipped instead of closed.
% The smooth profile samples the cubic falloff as opacity increments.
% 平滑模式将三次衰减离散成透明度增量，叠加后保持预定的不透明度。
\def\lg@shadowlayer#1#2#3#4#5{%
  \ifdim#1pt>0pt\relax
  \foreach\lg@i in{\lg@shadowsteps,...,1}{%
    \ifdefstring{\lg@shadowprofile}{smooth}{%
      \pgfmathsetmacro\lg@t{(\lg@i-.5)/\lg@shadowsteps}%
      \pgfmathsetlengthmacro\lg@dilation{max(#5+#2*\lg@t,-max(0,min(\lg@width,\lg@height)*.5-.05pt))}%
      \pgfmathsetmacro\lg@alpha{1-(1-#1)^(6*\lg@t*(1-\lg@t)/\lg@shadowsteps)}%
    }{%
      \pgfmathsetlengthmacro\lg@dilation{#5+#2*\lg@i/\lg@shadowsteps}%
      \pgfmathsetmacro\lg@alpha{min(1,#1*exp(-3*(\lg@i/\lg@shadowsteps)^2)/(\lg@shadowsteps*.49))}%
    }%
    \csname tcb@shadow@\tcb@split@state\endcsname
        {#3}{#4}{-\lg@dilation}%
        {overlay,fill=\lg@shadowcolor,draw=none,opacity=\lg@alpha}%
  }%
  \fi
}
\def\lg@shadowpaint{%
  \pgfmathsetmacro\lg@shadowtotal{\lg@shadow+\lg@contact}%
  \ifdim\lg@shadowtotal pt>0pt\relax
  \begin{scope}[overlay]
    \pgfmathsetlengthmacro\lg@shadowpad{max(abs(\lg@shadowx)+abs(\lg@shadowy)+\lg@shadowblur+abs(\lg@shadowspread),abs(\lg@contactx)+abs(\lg@contacty)+\lg@contactblur+abs(\lg@contactspread))+1mm}%
    \ifdefstring{\tcb@split@state}{first}{%
      \clip([xshift=-\lg@shadowpad]frame.south west)rectangle
                ([xshift=\lg@shadowpad,yshift=\lg@shadowpad]frame.north east);}{%
      \ifdefstring{\tcb@split@state}{middle}{%
        \clip([xshift=-\lg@shadowpad]frame.south west)rectangle
                  ([xshift=\lg@shadowpad]frame.north east);}{%
        \ifdefstring{\tcb@split@state}{last}{%
          \clip([xshift=-\lg@shadowpad,yshift=-\lg@shadowpad]frame.south west)rectangle
                    ([xshift=\lg@shadowpad]frame.north east);}{}%
      }%
    }%
    \begin{tcbinvclipframe}
      \lg@shadowlayer{\lg@shadow}{\lg@shadowblur}{\lg@shadowx}{\lg@shadowy}{\lg@shadowspread}%
      \lg@shadowlayer{\lg@contact}{\lg@contactblur}{\lg@contactx}{\lg@contacty}{\lg@contactspread}%
    \end{tcbinvclipframe}
  \end{scope}
  \fi
}
